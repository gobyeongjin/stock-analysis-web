"""Expanding-window walk-forward: A technical model, SWING_ATR, with/without KOSPI MA60.
Run at repository root: python backtest_walkforward_a.py
Requires existing backtest_lending_b2.py, backtest_market_filter_a.py,
analysis/backtest_atr_8pct.py, and CSV inputs.
"""
from pathlib import Path
import sys
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier

ROOT = Path(__file__).resolve().parent
ANALYSIS = ROOT / "analysis"
sys.path.insert(0, str(ANALYSIS))
import backtest_atr_8pct as engine
from backtest_lending_b2 import TECH, THRESHOLD
from backtest_market_filter_a import load_kospi

# Fixed, predeclared folds: 60%-70%, 70%-80%, 80%-90%, 90%-100%.
# Five trading-date embargo before each test fold.
START_FRACTION = 0.60
FOLDS = 4
PURGE_DATES = 5
STRATEGY = "SWING_ATR"


def load_technical_universe():
    """Keep the same 100-stock common universe as the earlier A/B2 comparison."""
    # Use existing loader to preserve exact eligibility, features and sample matching.
    # It returns the 80% split, so we combine both portions and recover the
    # 5 purged dates from the underlying source via a separate full loader below.
    from backtest_lending_b2 import LEND
    lending = pd.read_csv(ROOT / "lending_100.csv", dtype={"Ticker": str}, low_memory=False)
    lending["Code"] = lending["Ticker"].str.split(".").str[0].str.zfill(6)
    lending["dt"] = pd.to_datetime(lending["dt"], errors="coerce")
    lending["rmnd"] = pd.to_numeric(
        lending["rmnd"].astype(str).str.replace(",", "", regex=False), errors="coerce"
    )
    lending = (lending.dropna(subset=["dt", "rmnd"])
               .sort_values(["Code", "dt"])
               .drop_duplicates(["Code", "dt"], keep="last").copy())
    g = lending.groupby("Code", sort=False)["rmnd"]
    lending["Lending_Change_1D"] = g.pct_change(fill_method=None).replace([np.inf, -np.inf], np.nan) * 100
    lending["Lending_Change_5D"] = (lending["rmnd"] / g.shift(5) - 1).replace([np.inf, -np.inf], np.nan) * 100
    lending["Lending_Change_20D"] = (lending["rmnd"] / g.shift(20) - 1).replace([np.inf, -np.inf], np.nan) * 100
    lending["Lending_Up_Days_5D"] = (
        lending["Lending_Change_1D"].gt(0).astype(float)
        .groupby(lending["Code"]).transform(lambda s: s.rolling(5, min_periods=5).sum())
    )
    ma20 = g.transform(lambda s: s.rolling(20, min_periods=20).mean())
    lending["Lending_MA20_Gap"] = (lending["rmnd"] / ma20 - 1).replace([np.inf, -np.inf], np.nan) * 100
    lending[LEND] = lending.groupby("Code", sort=False)[LEND].shift(1)

    df = pd.read_csv(ANALYSIS / "training_data_v3.csv", dtype={"Ticker": str}, low_memory=False)
    df["Date"] = pd.to_datetime(df["Date"], errors="coerce")
    df["Code"] = df["Ticker"].str.split(".").str[0].str.zfill(6)
    df = df[df["Future_5D_Valid"].astype(str).str.strip().str.lower().isin(["true", "1", "1.0"])].copy()
    df = df.merge(
        lending[["Code", "dt"] + LEND], left_on=["Code", "Date"],
        right_on=["Code", "dt"], how="inner", validate="many_to_one"
    )
    df = df.replace([np.inf, -np.inf], np.nan).dropna(subset=TECH + LEND + ["Label", "Date"])
    df = df[df["Label"].isin([0, 1])].sort_values(["Date", "Ticker"]).reset_index(drop=True)
    return df


def fold_ranges(dates):
    n = len(dates)
    start = int(n * START_FRACTION)
    if start < PURGE_DATES + 30:
        raise ValueError("워크포워드 학습에 필요한 날짜가 부족합니다.")
    edges = np.linspace(start, n, FOLDS + 1, dtype=int)
    for i in range(FOLDS):
        a, b = int(edges[i]), int(edges[i + 1])
        if b <= a or a - PURGE_DATES <= 0:
            raise ValueError("워크포워드 분할 오류")
        yield i + 1, dates[a - PURGE_DATES], dates[a], dates[b - 1]


def main():
    df = load_technical_universe()
    dates = np.sort(df["Date"].unique())
    print(f"공통 표본 {len(df):,}행 / 종목 {df['Code'].nunique()}개 / 거래일 {len(dates)}개")
    print(f"임계값 {THRESHOLD}, 초기 자본 {engine.INITIAL_CASH:,.0f}원 (폴드마다 재설정)")

    prices_all = engine.load_prices()
    calendar = np.sort(prices_all["Date"].unique())
    kospi = load_kospi().sort_values("Date").copy()
    kospi["KOSPI_MA60"] = kospi["Close"].rolling(60, min_periods=60).mean()
    kospi = kospi.rename(columns={"Date": "Index_Date", "Close": "KOSPI_Close"})
    kospi = kospi[["Index_Date", "KOSPI_Close", "KOSPI_MA60"]]

    summaries, trades_out, equity_out = [], [], []
    for fold, train_cutoff, test_start, test_end in fold_ranges(dates):
        train = df[df["Date"] < train_cutoff]
        test = df[(df["Date"] >= test_start) & (df["Date"] <= test_end)]
        if train.empty or test.empty:
            continue
        print(f"\n[폴드 {fold}] 학습 {len(train):,}행, 평가 {len(test):,}행"
              f" | {pd.Timestamp(test_start).date()} ~ {pd.Timestamp(test_end).date()}", flush=True)
        model = RandomForestClassifier(
            n_estimators=200, max_depth=8, min_samples_leaf=10,
            class_weight="balanced", random_state=42, n_jobs=-1
        )
        model.fit(train[TECH], train["Label"].astype(int))
        signals = test[["Date", "Code"]].rename(columns={"Code": "Ticker"}).copy()
        signals["Probability"] = model.predict_proba(test[TECH])[:, 1]
        signals = signals[signals["Probability"] >= THRESHOLD].copy()

        # Allow five following trading dates to execute the last signal and exit.
        end_pos = np.searchsorted(calendar, np.datetime64(test_end), side="right")
        end_pos = min(end_pos + 4, len(calendar) - 1)
        end_price_date = calendar[end_pos]
        prices = prices_all[
            (prices_all["Date"] >= test_start) & (prices_all["Date"] <= end_price_date)
        ].copy()
        signals = signals.merge(
            prices[["Date", "Ticker", "ATR14"]],
            on=["Date", "Ticker"], how="left", validate="one_to_one"
        )
        signals = signals[signals["ATR14"].notna() & (signals["ATR14"] > 0)].copy()

        signals = pd.merge_asof(
            signals.sort_values("Date"), kospi.sort_values("Index_Date"),
            left_on="Date", right_on="Index_Date", direction="backward"
        )
        stale = (signals["Date"] - signals["Index_Date"]).dt.days > 7
        signals.loc[stale, ["KOSPI_Close", "KOSPI_MA60"]] = np.nan

        versions = {
            "BASE_A": signals,
            "KOSPI_MA60": signals[signals["KOSPI_Close"] > signals["KOSPI_MA60"]],
        }
        for name, selected in versions.items():
            s = selected[["Date", "Ticker", "Probability", "ATR14"]].copy()
            print(f"  {name}: 신호 {len(s):,}건, 백테스트 중...", flush=True)
            trades, equity = engine.run_backtest_v4(prices, s, STRATEGY)
            result = engine.summarize_results(trades, equity, STRATEGY)
            result.update({
                "Fold": fold, "Filter": name,
                "Train_End": (pd.Timestamp(train_cutoff) - pd.Timedelta(days=1)).date(),
                "Test_Start": pd.Timestamp(test_start).date(),
                "Test_End": pd.Timestamp(test_end).date(),
                "Train_Rows": len(train), "Test_Rows": len(test),
                "Signal_Count": len(s), "Threshold": THRESHOLD,
            })
            summaries.append(result)
            if not trades.empty:
                trades = trades.copy()
                trades.insert(0, "Fold", fold)
                trades.insert(1, "Filter", name)
                trades_out.append(trades)
            if not equity.empty:
                equity = equity.copy()
                equity.insert(0, "Fold", fold)
                equity.insert(1, "Filter", name)
                equity_out.append(equity)

    if not summaries:
        raise RuntimeError("평가 결과가 없습니다. 입력 데이터를 확인하세요.")

    result = pd.DataFrame(summaries)
    summary_path = ANALYSIS / "walkforward_a_summary.csv"
    result.to_csv(summary_path, index=False, encoding="utf-8-sig")
    if trades_out:
        pd.concat(trades_out, ignore_index=True).to_csv(
            ANALYSIS / "walkforward_a_trades.csv", index=False, encoding="utf-8-sig"
        )
    if equity_out:
        pd.concat(equity_out, ignore_index=True).to_csv(
            ANALYSIS / "walkforward_a_equity.csv", index=False, encoding="utf-8-sig"
        )
    columns = ["Fold", "Filter", "Test_Start", "Test_End",
               "Signal_Count", "Return_Pct", "MDD_Pct", "Trades", "Win_Rate_Pct"]
    print("\n=== A 모델 워크포워드 결과 ===")
    print(result[columns].to_string(index=False))

    # Equal-weight arithmetic fold mean, NOT an investable continuous equity curve.
    overview = (result.groupby("Filter", as_index=False)
                .agg(Avg_Return_Pct=("Return_Pct", "mean"),
                     Avg_MDD_Pct=("MDD_Pct", "mean"),
                     Positive_Folds=("Return_Pct", lambda x: int((x > 0).sum())),
                     Total_Trades=("Trades", "sum")))
    overview[["Avg_Return_Pct", "Avg_MDD_Pct"]] = overview[
        ["Avg_Return_Pct", "Avg_MDD_Pct"]
    ].round(2)
    overview.to_csv(ANALYSIS / "walkforward_a_overview.csv", index=False, encoding="utf-8-sig")
    print("\n=== 폴드별 단순 평균 (연속 복리 수익률 아님) ===")
    print(overview.to_string(index=False))
    print("\n저장:", summary_path)
    print("주의: 기존에 탐색한 구간을 포함하므로 독립적인 최종 OOS 검증은 아닙니다.")
    print("주의: 폴드마다 초기 자본을 재설정합니다. 원본 엔진의 체결/비용 가정을 유지합니다.")


if __name__ == "__main__":
    main()
